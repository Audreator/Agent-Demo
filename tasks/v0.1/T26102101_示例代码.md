## 1. MCP stdio 客户端骨架（JSON-RPC 2.0）

要点：一行一条 JSON；stdout 只读协议；日志走 stderr；按 id 匹配响应。

```python
import asyncio, json, itertools, os

class MCPClient:
    def __init__(self, cmd: list[str], cwd: str | None = None, timeout: float = 30.0):
        self._cmd, self._cwd, self._timeout = cmd, cwd, timeout
        self._ids = itertools.count(1)
        self._proc: asyncio.subprocess.Process | None = None
        self._pending: dict[int, asyncio.Future] = {}
        self._reader_task: asyncio.Task | None = None

    async def start(self):
        self._proc = await asyncio.create_subprocess_exec(
            *self._cmd, cwd=self._cwd,
            stdin=asyncio.subprocess.PIPE,
            stdout=asyncio.subprocess.PIPE,
            stderr=asyncio.subprocess.PIPE,   # 日志必须走 stderr
        )
        self._reader_task = asyncio.create_task(self._read_loop())
        await self.call("initialize", {
            "protocolVersion": "2024-11-05",
            "capabilities": {},
            "clientInfo": {"name": "agent-client", "version": "0.1"},
        })
        await self.notify("notifications/initialized", {})

    async def _read_loop(self):
        assert self._proc and self._proc.stdout
        while line := await self._proc.stdout.readline():
            line = line.strip()
            if not line:
                continue
            try:
                msg = json.loads(line)
            except json.JSONDecodeError:
                continue          # 非协议行直接丢弃
            if "id" in msg and msg["id"] in self._pending:
                fut = self._pending.pop(msg["id"])
                if not fut.done():
                    fut.set_result(msg)
            # 否则视为服务端通知，交给 handler

    async def call(self, method: str, params: dict) -> dict:
        rid = next(self._ids)
        fut = asyncio.get_running_loop().create_future()
        self._pending[rid] = fut
        payload = {"jsonrpc": "2.0", "id": rid, "method": method, "params": params}
        assert self._proc and self._proc.stdin
        self._proc.stdin.write((json.dumps(payload, ensure_ascii=False) + "\n").encode())
        await self._proc.stdin.drain()
        try:
            msg = await asyncio.wait_for(fut, timeout=self._timeout)
        finally:
            self._pending.pop(rid, None)
        if "error" in msg:
            raise MCPError(msg["error"].get("code"), msg["error"].get("message"))
        return msg["result"]

    async def notify(self, method: str, params: dict):
        payload = {"jsonrpc": "2.0", "method": method, "params": params}
        assert self._proc and self._proc.stdin
        self._proc.stdin.write((json.dumps(payload, ensure_ascii=False) + "\n").encode())
        await self._proc.stdin.drain()

    async def list_tools(self) -> list[dict]:
        return (await self.call("tools/list", {}))["tools"]

    async def call_tool(self, name: str, arguments: dict) -> dict:
        return await self.call("tools/call", {"name": name, "arguments": arguments})
```

TOOLS: dict[str, dict] = {}

def register(name: str, description: str, input_schema: dict):
    def deco(fn):
        TOOLS[name] = {"name": name, "description": description,
                       "inputSchema": input_schema, "_fn": fn}

async def handle(msg: dict) -> dict | None:
    method, rid = msg.get("method"), msg.get("id")
    if method == "initialize":
        return ok(rid, {"protocolVersion": "2024-11-05",
                        "capabilities": {"tools": {}},
                        "serverInfo": {"name": "agent-tools", "version": "0.1"}})
    if method == "tools/list":
        return ok(rid, {"tools": [
            {"name": t["name"], "description": t["description"],
             "inputSchema": t["inputSchema"]} for t in TOOLS.values()
        ]})
    if method == "tools/call":
        name = msg["params"]["name"]
        args = msg["params"].get("arguments") or {}
        try:
            result = await TOOLS[name]["_fn"](**args)
            return ok(rid, {"content": [{"type": "text", "text": json_dumps(result)}],
                            "isError": False})
        except Exception as exc:
            return ok(rid, {"content": [{"type": "text", "text": f"{type(exc).__name__}: {exc}"}],

## 3. ReAct 输出协议与容错解析

```python
import json, re

FIELD = re.compile(r"(Thought|Action|Action Input|Observation|Final Answer)\s*[:：]", re.I)

def split_fields(text: str) -> dict[str, str]:
    parts, last_key, buf = {}, None, []
    for line in text.splitlines():
        m = re.match(r"^\s*(Thought|Action|Action Input|Final Answer)\s*[:：]\s*(.*)$", line, re.I)
        if m:
            if last_key:
                parts[last_key] = "\n".join(buf).strip()
            last_key, buf = m.group(1).lower(), [m.group(2)]
        elif last_key:
            buf.append(line)
    if last_key:
        parts[last_key] = "\n".join(buf).strip()
    return parts

def repair_json(raw: str) -> dict | None:
    raw = re.sub(r"^```(?:json)?|```$", "", raw.strip(), flags=re.M).strip()
    try:
        return json.loads(raw)
    except json.JSONDecodeError:
        pass
    start = raw.find("{")
    if start == -1:
        return None
    depth = 0
    for i in range(start, len(raw)):
        depth += raw[i] == "{"
        depth -= raw[i] == "}"
        if depth == 0:
            chunk = raw[start:i + 1]
            chunk = re.sub(r",\s*}", "}", chunk)
            try:
                return json.loads(chunk)
            except json.JSONDecodeError:
                return None
    return None

def parse(text: str, registry: dict[str, dict]) -> dict:
    """返回 {kind: 'final'|'action'|'invalid', ...}"""
    f = split_fields(text)
    if "final answer" in f:
        return {"kind": "final", "thought": f.get("thought", ""), "answer": f["final answer"]}
    if "action" in f:
        name = f["action"].strip().strip("`'\"")
        if name not in registry:
            return {"kind": "invalid", "reason": f"unknown tool: {name}"}
        args = repair_json(f.get("action input", ""))
        if args is None:
            return {"kind": "invalid", "reason": "action input is not valid JSON"}
        return {"kind": "action", "thought": f.get("thought", ""), "name": name, "arguments": args}
    return {"kind": "invalid", "reason": "neither Action nor Final Answer found"}
```

def normalize_native(resp: dict, registry: dict[str, dict]) -> dict:
    choice = resp["choices"][0]["message"]
    reasoning = choice.get("reasoning_content") or ""
    calls = choice.get("tool_calls") or []
    if not calls:
        return {"kind": "final", "thought": reasoning, "answer": choice.get("content") or ""}
    call = calls[0]
    name = call["function"]["name"]
    args = json.loads(call["function"].get("arguments") or "{}")
    if name not in registry:
        return {"kind": "invalid", "reason": f"unknown tool: {name}"}
    return {"kind": "action", "thought": reasoning, "name": name,
            "arguments": args, "call_id": call.get("id")}

## 6. 熔断器状态机

```python
import time

class Breaker:
    def __init__(self, open_seconds: int = 30, retries: int = 3):
        self.open_seconds, self.retries = open_seconds, retries
        self.state = "closed"            # closed | open | half_open
        self.opened_at = 0.0
        self._probe_inflight = False

    def allow(self) -> bool:
        if self.state == "closed":
            return True
        if self.state == "open":
            if time.monotonic() - self.opened_at >= self.open_seconds:
                self.state, self._probe_inflight = "half_open", False
            else:
                return False
        if self.state == "half_open":
            if self._probe_inflight:
                return False             # 半开只放行 1 个探测
            self._probe_inflight = True
            return True
        return False

    def on_success(self):
        self.state, self._probe_inflight = "closed", False

    def on_failure(self):
        self.state, self.opened_at, self._probe_inflight = "open", time.monotonic(), False
```

## 8. calculator 安全求值（AST 白名单）

```python
import ast
from decimal import Decimal, getcontext

ALLOWED = (ast.Expression, ast.BinOp, ast.UnaryOp, ast.Constant,
           ast.Add, ast.Sub, ast.Mult, ast.Div, ast.Pow, ast.Mod,
           ast.USub, ast.UAdd)

def safe_eval(expr: str, precision: int = 28) -> Decimal:
    getcontext().prec = precision
    tree = ast.parse(expr, mode="eval")
    for node in ast.walk(tree):
        if not isinstance(node, ALLOWED):
            raise ValueError(f"不允许的语法: {type(node).__name__}")
        if isinstance(node, ast.Constant) and not isinstance(node.value, (int, float)):
            raise ValueError("只允许数值常量")
    def _ev(n):
        if isinstance(n, ast.Expression):
            return _ev(n.body)
        if isinstance(n, ast.Constant):
            return Decimal(str(n.value))
        if isinstance(n, ast.UnaryOp):
            v = _ev(n.operand)
            return +v if isinstance(n.op, ast.UAdd) else -v
        a, b = _ev(n.left), _ev(n.right)
        if isinstance(n.op, ast.Add): return a + b
        if isinstance(n.op, ast.Sub): return a - b
        if isinstance(n.op, ast.Mult): return a * b
        if isinstance(n.op, ast.Div):
            if b == 0: raise ZeroDivisionError("除数不能为 0")
            return a / b
        if isinstance(n.op, ast.Mod): return a % b
        if isinstance(n.op, ast.Pow):
            if b != int(b) or abs(b) > 1000: raise ValueError("指数需为不超过 1000 的整数")
            return a ** int(b)
        raise ValueError("不支持的运算符")
    return _ev(tree)
```
