```python
import re

REFUSAL_PATTERNS = (
    "我不知道",

    "i do not know",
    "i'm not sure",
    "i am not sure",
    "no information",
    "cannot answer",
    "can't answer",
    "unable to answer",
)

REFUSAL_RE = re.compile("|".join(re.escape(item) for item in REFUSAL_PATTERNS), re.IGNORECASE)
URL_RE = re.compile(r"https?://", re.IGNORECASE)


def is_refusal_answer(
    answer: str, *, match_window: int = 200, max_answer_len: int = 800
) -> bool:
    text = (answer or "").strip()
    if not text or len(text) > max_answer_len:
        return False
    if URL_RE.search(text):
        return False
    return bool(REFUSAL_RE.search(text[:match_window]))


def should_force_search(
    *,
    answer: str,
    has_ok_search: bool,
    already_forced: bool,
    enabled: bool,
    finish_kind: str,
) -> bool:
    if not enabled or already_forced or has_ok_search:
        return False
    if finish_kind not in {"final", "final_answer"}:
        return False
    return is_refusal_answer(answer)
```

要点：

- 判定只针对最终回答，不针对思考、工具调用、收束总结、解析错误。
- `has_ok_search` 取自本轮 `tool_events` 中是否存在 `name == "search" and status == "ok"`。
- 长度、链接、匹配窗口三个条件是防误判的关键，不要省略。

## 2. 最终回答缓冲

改造现有 `StreamSplitter`（示意）：

```python
class StreamSplitter:
    def __init__(self, mode: str, emit: Emit, *, buffer_final: bool = True) -> None:
        self.mode = mode
        self.emit = emit
        self.buffer_final = buffer_final
        self.pending = ""
        self.final_started = False
        self.final_buffer = ""
        self.buffered_chars = 0
        self.buffer_limit_hit = False
        self.buffer_limit = 20000
        self.message_chars = 0
        self.thought_chars = 0

    async def _final(self, text: str) -> None:
        self.message_chars += len(text)
        if not self.buffer_final or self.buffer_limit_hit:
            await self.emit("message_delta", {"delta": text})
            return
        if self.buffered_chars + len(text) > self.buffer_limit:
            self.buffer_limit_hit = True
            pending = self.final_buffer + text
            self.final_buffer = ""
            self.buffered_chars = 0
            await self.emit("message_delta", {"delta": pending})
            return
        self.final_buffer += text
        self.buffered_chars += len(text)

    async def feed(self, chunk: ChatChunk) -> None:
        if chunk.reasoning:
            self.thought_chars += len(chunk.reasoning)
            await self.emit("thought_delta", {"delta": chunk.reasoning})
        if not chunk.content:
            return
        if self.mode == "native":
            await self._final(chunk.content)
            return
        if self.final_started:
            await self._final(chunk.content)
            return
        self.pending += chunk.content
        match = FINAL_ANSWER_RE.search(self.pending)
        if match:
            before = self.pending[: match.start()]
            after = self.pending[match.end() :]
            if before.strip():
                self.thought_chars += len(before)
                await self.emit("thought_delta", {"delta": before})
            self.final_started = True
            self.pending = ""
            if after:
                await self._final(after)
            return
        keep = 24
        if len(self.pending) > keep:
            safe = self.pending[:-keep]
            self.pending = self.pending[-keep:]
            if safe.strip():
                self.thought_chars += len(safe)
                await self.emit("thought_delta", {"delta": safe})

    def take_final(self) -> str:
        text, self.final_buffer = self.final_buffer, ""
        self.buffered_chars = 0
        return text

    def buffered_too_long(self, limit: int) -> bool:
        return self.buffer_limit_hit
```

补充：

- 超过 `FINAL_ANSWER_BUFFER_MAX` 时置 `buffer_limit_hit = True`，主循环立即放行并跳过判定。
- `reset()` 必须清空 `pending`、`final_buffer`、`final_started` 与超限标记。
- `flush()` 在收束流程中应保持原有直推行为，避免收束总结被缓冲。

```python
async def _run_forced_search(self, state: TurnState, emit: Emit) -> Any:
    query = state.user_input.strip()[: self.settings.forced_search_query_max]
    tool_call_id = uuid.uuid4().hex
    await emit(
        "tool_start",
        {
            "tool_call_id": tool_call_id,
            "name": "search",
            "arguments": {"query": query, "max_results": 5},
            "turn_no": state.turn_no,
            "forced": True,
        },
    )
    started = time.monotonic()
    outcome = await self.executor.execute(
        "search",
        {"query": query, "max_results": 5},
        session_id=state.session.id,
        turn_no=state.turn_no,
        tool_call_id=tool_call_id,
    )
    latency = int((time.monotonic() - started) * 1000)
    await self.tracer.emit_event(
        "search_forced",
        {
            "trigger": "uncertainty",
            "query_chars": len(query),
            "status": "ok" if outcome.ok else "error",
            "result_count": outcome.result_count,
            "reachable_count": outcome.reachable_count,
            "latency_ms": latency,
        },
    )
    return outcome
```

## 5. 轮次首可见采集

```python
def wrap_emit(emit: Emit, turn_span: SpanInfo, started: float) -> Emit:
    seen = False

    async def inner(kind: str, payload: dict[str, Any]) -> None:
        nonlocal seen
        if not seen and kind in {"thought_delta", "message_delta"}:
            seen = True
            turn_span.set(first_token_ms=int((time.monotonic() - started) * 1000))
        await emit(kind, payload)

    return inner
```

在 `run()` 中进入 `turn` span 后，用包装后的 `emit` 传入 `_run()`；`turn` span 结束时会把字段写入 `turn` 事件。

```python
import pytest
from app.agent.uncertainty import is_refusal_answer, should_force_search


@pytest.mark.parametrize(
    ("answer", "expected"),
    [
        ("我不知道这个问题的答案。", True),
        ("很抱歉，我无法确定。", True),
        ("I'm not sure about that.", True),
        ("根据检索结果，2025 年营收为 120 亿元。https://example.com", False),
        ("不确定的部分我已经在第三段说明了全部结论。", False),
        ("长" * 900 + "我不知道", False),
    ],
)
def test_is_refusal_answer(answer: str, expected: bool) -> None:
    assert is_refusal_answer(answer) is expected


def test_should_force_search_skips_when_search_ok() -> None:
    assert not should_force_search(
        answer="我不知道",
        has_ok_search=True,
        already_forced=False,
        enabled=True,
        finish_kind="final",
    )
```

集成测试构造两种 MCP 桩：返回空结果、返回错误；断言用户侧最终消息为固定文案，且 `tool_calls` 中存在一条 `search`，事件中存在 `search_forced`。后台指标用构造的 `turn` / `model_success` 事件断言 P95 与样本数。
