## 2. SSRF 安全链接校验

```python
from __future__ import annotations

import asyncio
import ipaddress
import socket
from dataclasses import dataclass
from urllib.parse import urljoin, urlparse

import httpx

BLOCKED_NETS = [
    ipaddress.ip_network("127.0.0.0/8"),
    ipaddress.ip_network("10.0.0.0/8"),
    ipaddress.ip_network("172.16.0.0/12"),
    ipaddress.ip_network("192.168.0.0/16"),
    ipaddress.ip_network("169.254.0.0/16"),
    ipaddress.ip_network("::1/128"),
    ipaddress.ip_network("fc00::/7"),
    ipaddress.ip_network("fe80::/10"),
]
ALLOWED_PORTS = {80, 443}


@dataclass
class LinkStatus:
    url: str
    reachable: bool
    status_code: int | None
    reason: str = ""


def _is_public_ip(value: str) -> bool:
    try:
        ip = ipaddress.ip_address(value)
    except ValueError:
        return False
    return not any(ip in net for net in BLOCKED_NETS)


async def _check_host(host: str) -> bool:
    try:
        # DNS 解析放线程池，避免阻塞事件循环
        infos = await asyncio.to_thread(socket.getaddrinfo, host, None)
    except socket.gaierror:
        return False
    return all(_is_public_ip(info[4][0]) for info in infos)


async def _validate_url(url: str) -> str | None:
    parsed = urlparse(url)
    if parsed.scheme not in {"http", "https"}:
        return "仅允许 http/https"
    if not parsed.hostname or not await _check_host(parsed.hostname):
        return "域名解析到私网或不可解析"
    port = parsed.port or (443 if parsed.scheme == "https" else 80)
    if port not in ALLOWED_PORTS:
        return "端口不允许"
    return None


async def check_link(url: str, timeout: float = 8.0) -> LinkStatus:
    """手动跟随重定向，每一跳都做 SSRF 校验。"""
    current = url
    async with httpx.AsyncClient(timeout=timeout) as client:
        for _ in range(5):
            problem = await _validate_url(current)
            if problem:
                return LinkStatus(current, False, None, problem)
            try:
                response = await client.head(current, follow_redirects=False)
                if response.status_code in {400, 403, 405, 429}:
                    response = await client.get(
                        current,
                        headers={"Range": "bytes=0-2048"},
                        follow_redirects=False,
                    )
            except httpx.HTTPError as exc:
                return LinkStatus(current, False, None, f"请求失败: {exc}")
            if response.status_code in {301, 302, 303, 307, 308}:
                location = response.headers.get("location")
                if not location:
                    return LinkStatus(current, False, response.status_code, "重定向缺少 Location")
                current = urljoin(current, location)
                continue
            reachable = 200 <= response.status_code < 400
            return LinkStatus(url, reachable, response.status_code)
    return LinkStatus(url, False, None, "重定向次数过多")
```

## 3. 引用归并

```python
from urllib.parse import urlsplit, urlunsplit


def normalize_url(url: str) -> str:
    parts = urlsplit(url.strip())
    return urlunsplit(
        (parts.scheme.lower(), parts.netloc.lower(), parts.path, parts.query, "")
    )


def merge_citations(groups: list[list[dict]], limit: int = 10) -> list[dict]:
    merged: list[dict] = []
    seen: set[str] = set()
    for group in groups:
        for item in group:
            url = str(item.get("url") or "").strip()
            if not url:
                continue
            key = normalize_url(url)
            if key in seen:
                continue
            seen.add(key)
            merged.append(
                {
                    "id": f"c{len(merged) + 1}",
                    "title": str(item.get("title") or "")[:200],
                    "url": key,
                    "snippet": str(item.get("snippet") or "")[:300],
                    "reachable": False,
                    "status_code": None,
                    "checked_at": None,
                }
            )
            if len(merged) >= limit:
                return merged
    return merged
```

## 4. search 工具集成

```python
import asyncio
from datetime import UTC, datetime


async def search_tool_with_links(query: str, max_results: int = 5, **kwargs) -> dict:
    results = await provider.search(query, max_results=max_results)
    citations = merge_citations([results], limit=10)
    checks = await asyncio.gather(
        *(check_link(item["url"]) for item in citations),
        return_exceptions=True,
    )
    for item, checked in zip(citations, checks):
        if isinstance(checked, BaseException):
            continue
        item["reachable"] = checked.reachable
        item["status_code"] = checked.status_code
        item["checked_at"] = datetime.now(UTC).isoformat()
    # 校验失败不抛出，只带 reachable=False 返回
    return {
        "query": query,
        "results": citations,
        "reachable_count": sum(1 for item in citations if item["reachable"]),
    }
```

## 5. 评价 upsert

```python
from sqlalchemy.dialects.postgresql import insert
from app.models.db_models import MessageFeedback, utcnow


async def upsert_feedback(db, *, message_id, session_id, user_id, rating, reason_code, reason_text, trace_id):
    stmt = (
        insert(MessageFeedback)
        .values(
            message_id=message_id,
            session_id=session_id,
            user_id=user_id,
            rating=rating,
            reason_code=reason_code,
            reason_text=reason_text,
            trace_id=trace_id,
            created_at=utcnow(),
            updated_at=utcnow(),
        )
        .on_conflict_do_update(
            constraint="uq_feedback_message_user",
            set_={
                "rating": rating,
                "reason_code": reason_code,
                "reason_text": reason_text,
                "trace_id": trace_id,
                "updated_at": utcnow(),
            },
        )
    )
    await db.execute(stmt)
    await db.commit()
```

## 6. 后台指标查询

```python
from sqlalchemy import case, func, select
from app.models.db_models import ToolCall


async def tool_metrics(db, start, end):
    ok = func.count().filter(ToolCall.status == "ok")
    error = func.count().filter(ToolCall.status == "error")
    blocked = func.count().filter(ToolCall.status == "blocked")
    p95 = func.percentile_cont(0.95).within_group(ToolCall.latency_ms.asc())
    stmt = (
        select(
            ToolCall.tool_name,
            func.count().label("calls"),
            ok.label("ok"),
            error.label("error"),
            blocked.label("blocked"),
            (ok * 1.0 / func.nullif(ok + error, 0)).label("success_rate"),
            func.avg(ToolCall.latency_ms).label("avg_ms"),
            p95.label("p95_ms"),
        )
        .where(ToolCall.created_at >= start, ToolCall.created_at < end)
        .group_by(ToolCall.tool_name)
    )
    return (await db.execute(stmt)).mappings().all()
```

## 7. CSV 导出

```python
import csv
import io

FORMULA_PREFIXES = ("=", "+", "-", "@")


def safe_cell(value) -> str:
    text = "" if value is None else str(value)
    if text.startswith(FORMULA_PREFIXES):
        return "'" + text
    return text


def build_csv(rows: list[dict]) -> bytes:
    buffer = io.StringIO()
    writer = csv.DictWriter(buffer, fieldnames=list(rows[0].keys()) if rows else ["turn_no"])
    writer.writeheader()
    for row in rows:
        writer.writerow({key: safe_cell(value) for key, value in row.items()})
    return ("\ufeff" + buffer.getvalue()).encode("utf-8")
```

import pytest


@pytest.mark.parametrize(
    "url, expected",
    [
        ("http://127.0.0.1/a", False),

        ("file:///etc/passwd", False),

async def test_link_ssrf_guard(url, expected):
    status = await check_link(url, timeout=0.5)
    assert status.reachable is expected


async def test_feedback_upsert_is_idempotent(db, message_factory):
    message = await message_factory(role="assistant", meta={"kind": "final"})
    first = await upsert_feedback(db, message_id=message.id, session_id=message.session_id,
                                  user_id="dev-user", rating="like",
                                  reason_code=None, reason_text=None, trace_id=None)
    second = await upsert_feedback(db, message_id=message.id, session_id=message.session_id,
                                   user_id="dev-user", rating="dislike",
                                   reason_code="too_slow", reason_text=None, trace_id=None)
    rows = await db.execute(select(MessageFeedback).where(MessageFeedback.message_id == message.id))
